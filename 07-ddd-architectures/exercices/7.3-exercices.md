*Niveau 3. Faire passer un service par ses ports ; les tests sont

```go
// Reminders sends the reminders. Now and Notify are the ports to the
// clock and to the messages; today the code does not use them.
type Reminders struct {
    Now    func() time.Time
    Notify app.Notifier
```

```go
// Send reminds each member who has not paid, if now is the day before the
// round, from 18:00. It returns how many reminders went out.
func (r Reminders) Send(ctx context.Context, round Round) int {
    now := time.Now()
    eve := round.Date.AddDate(0, 0, -1).Add(18 * time.Hour)
    if now.Before(eve) || !now.Before(round.Date) {
        return 0
    }
    sent := 0
    for _, phone := range round.Unpaid {
        form := url.Values{"to": {phone}, "text": {fmt.Sprintf("Rappel : demain, cotisation de %d FCFA", round.ShareFCFA)}}
        resp, err := http.Post("https://sms.gateway.example/send", "application/x-www-form-urlencoded", strings.NewReader(form.Encode()))
        if err == nil {
            resp.Body.Close()
            sent++
        }
    }
    return sent
}
```

## Exercice 5 — Incident : l'« upsert » du mardi

*Niveau 7. Un incident à mesurer, puis à empêcher.*

Le mardi 10 novembre 2026, un développeur « simplifie » l'enregistrement
d'un tour en une seule instruction :

```go
// Save writes r.
func (s Upsert) Save(ctx context.Context, id string, r *round.Round, version int) error {
    state, err := json.Marshal(r.Snapshot())
    if err != nil {
        return err
    }
    _, err = s.Pool.Exec(ctx, `INSERT INTO round (id, version, state) VALUES ($1, $2 + 1, $3)
        ON CONFLICT (id) DO UPDATE SET state = EXCLUDED.state, version = round.version + 1`, id, version, state)
    return err
}
```

Tous les tests passent : ceux de l'application tournent sur l'adaptateur
mémoire. Le soir, les trésorières voient des tours où presque personne
n'a payé, alors que chaque membre a reçu « Cotisation reçue ».

```go
// Evening replays the evening of Tuesday 10 November: 50 rounds of 20
// members, every member paying at about the same time, through the
// application and the store given by newStore. It returns the answers, and
// the members each stored round holds as paid.
func Evening(newStore func(pgstore.Rounds) app.Rounds) ([]Answer, map[string][]string, error) {
    ctx := context.Background()
    pool, drop, err := pg.Fresh(ctx, pgstore.Schema)
    if err != nil {
        return nil, nil, err
    }
    defer drop()
    store := newStore(pgstore.Rounds{Pool: pool})
    s := &app.Service{Rounds: store, Notify: &sms.Recorder{}}
    var members []string
    for i := range 20 {
        members = append(members, fmt.Sprintf("+2376770%05d", i))
    }
    var mu sync.Mutex
    var answers []Answer
    var wg sync.WaitGroup
    for n := range 50 {
        id := fmt.Sprintf("t%02d-r7", n)
        if err := store.Create(ctx, id, round.New(7, 10_000, members)); err != nil {
            return nil, nil, err
        }
        for _, m := range members {
            wg.Go(func() {
                err := s.Contribute(ctx, id, m, 10_000)
                mu.Lock()
                defer mu.Unlock()
                answers = append(answers, Answer{id, m, err == nil})
            })
        }
    }
    wg.Wait()
    paid := map[string][]string{}
    for n := range 50 {
        id := fmt.Sprintf("t%02d-r7", n)
        r, _, err := store.Load(ctx, id)
        if err != nil {
            return nil, nil, err
        }
        paid[id] = r.Paid()
    }
    return answers, paid, nil
}
```
