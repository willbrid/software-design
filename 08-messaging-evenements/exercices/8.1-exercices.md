```go

```

```go
// Close orders the payout and, once it is done, announces RoundClosed; a
// refused payout fails the closure and announces nothing. A notifier
// that is down does not fail the closure.
//
// Today Close calls every notifier itself, and any failure fails it.
func (t *Tontine) Close(ctx context.Context, round, beneficiary string, pot int64) error {
    if err := t.Payer.PayOut(ctx, round, beneficiary, pot); err != nil {
        return fmt.Errorf("pay out: %w", err)
    }
    for _, n := range t.Notifiers {
        if err := n.RoundClosed(ctx, round, beneficiary, pot); err != nil {
            return fmt.Errorf("notify: %w", err)
        }
    }
    return nil
}
```

## Exercice 5 — Incident : la semaine sans cagnottes

*Niveau 7. Un incident à mesurer, puis à rejouer comme il aurait dû
être fait.*

Le mercredi 18 novembre 2026 à 10 h, l'équipe des tontines renomme son
événement `RoundClosed` en `RoundCompleted`, un nom qu'elle trouve plus
clair. Les paiements écoutent toujours `RoundClosed`. Le dimanche soir, une
bénéficiaire appelle : elle attend sa cagnotte depuis mercredi.

```go
// Week closes 8 rounds a day, at 8:00, 9:30, ... , from Monday to
// Sunday, and returns what the tontines closed and what the payments
// paid. The tontines see no error.
func Week() (closed []Closure, paid []Payout) {
    b := bus.New(time.Millisecond)
    defer b.Close()
    var mu sync.Mutex
    b.Subscribe("RoundClosed", "payments", func(_ context.Context, m bus.Message) error {
        c := m.(Closed)
        mu.Lock()
        defer mu.Unlock()
        paid = append(paid, Payout{Round: c.Round, Pot: c.Pot, At: c.At})
        return nil
    })
    for day := range 7 {
        for i := range 8 {
            at := Monday.AddDate(0, 0, day).Add(8*time.Hour + time.Duration(i)*90*time.Minute)
            n := day*8 + i
            c := Closure{
                Round:       fmt.Sprintf("r%02d", n),
                Beneficiary: fmt.Sprintf("m%02d", n%25),
                Pot:         int64(10+5*(n%4)) * 10000,
                At:          at,
            }
            name := "RoundClosed"
            if !at.Before(Rename) {
                name = "RoundCompleted"
            }
            b.Publish(Closed{Name: name, Closure: c})
            closed = append(closed, c)
        }
    }
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()
    if err := b.Drain(ctx); err != nil {
        panic(err)
    }
    mu.Lock()
    defer mu.Unlock()
    return closed, paid
}
```
