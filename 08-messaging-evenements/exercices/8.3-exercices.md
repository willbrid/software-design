```go
// Sunday delivers the day to c: at 16:00, c restarts and the broker
// delivers again the events from 14:00.
func Sunday(c Consumer) {
    day := Day()
    replayed := false
    for i := 0; i < len(day); i++ {
        if day[i].Hour == 16 && !replayed {
            replayed = true
            c.Restart()
            for j := range day[:i] {
                if day[j].Hour >= 14 {
                    c.Handle(day[j])
                }
            }
        }
        c.Handle(day[i])
    }
}
```
