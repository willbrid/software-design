```go
    Status string // "En cours", "Remboursé", or "Autre"
    Left   string // what is left to repay, "100 000 FCFA", or "—"

```

```go
// ReadLoan reads the answer as the application does today.
func ReadLoan(body []byte) (LoanView, error) {
    var l struct {
        Member     string `json:"member"`
        AmountFCFA int64  `json:"amount_fcfa"`
        Status     string `json:"status"`
        RepaidFCFA int64  `json:"repaid_fcfa"`
    }
    if err := json.Unmarshal(body, &l); err != nil {
        return LoanView{}, err
    }
    var status string
    switch l.Status {
    case "open":
        status = "En cours"
    case "repaid":
        status = "Remboursé"
    default:
        return LoanView{}, fmt.Errorf("unknown status %q", l.Status)

    return LoanView{l.Member, fcfa(l.AmountFCFA), status, fcfa(l.AmountFCFA - l.RepaidFCFA)}, nil
}
```

## Exercice 3 — Refactoring : faire évoluer la réponse sans casser

*Niveau 3. Réécrire l'encodeur de la réponse ; les tests sont fournis.*

L'équipe veut trois choses dans la réponse de `GET /tontines/{id}` :

- l'identifiant en texte, `id_str`, pour les navigateurs (leçon 6.2) ;
- la part avec sa devise, `share: {"amount": 10000, "currency": "XAF"}` ;
- les membres paginés, `members_page: {"items": [...50 au plus],
  "next": <curseur ou null>}`.

```go
// Encode writes the answer for t, to the application client (its
// User-Agent, as "strict/2.0" or "plain/3.1").
func Encode(t Tontine, client string) []byte {
    b, _ := json.Marshal(struct {
        ID        int64    `json:"id"`
        Name      string   `json:"name"`
        ShareFCFA int64    `json:"share_fcfa"`
        Status    string   `json:"status"`
        NextRound int      `json:"next_round"`
        Members   []Member `json:"members"`
    }{t.ID, t.Name, t.ShareFCFA, t.Status, t.NextRound, t.Members})
    return b
}
```

## Exercice 5 — Incident : le statut `late`

*Niveau 7. Un incident à mesurer, puis à corriger côté serveur.*

Le lundi 2 novembre 2026 à 6 h, le serveur commence à renvoyer le statut
`late` pour les prêts en retard. Les versions 3.1 à 3.3 de l'application
lisent ainsi la liste des prêts :

```go
// ReadBefore34 is the reader of the versions 3.1 to 3.3: it knows open and
// repaid, and fails on any other status.
func ReadBefore34(body []byte) ([]string, error) {
    var a struct{ Loans []Loan }
    if err := json.Unmarshal(body, &a); err != nil {
        return nil, err
    }
    var out []string
    for _, l := range a.Loans {
        if l.Status != "open" && l.Status != "repaid" {
            return nil, fmt.Errorf("unknown status %q", l.Status)
        }
        out = append(out, labels[l.Status])
    }
    return out, nil
}
```

La version 3.4, qui connaît `late`, sort le mercredi 4 ; les membres
l'installent peu à peu. `ex5/week.go` rejoue la semaine : 4 000 membres
qui ont un prêt ouvrent l'écran à peu près une fois par jour.

```go
// Session is one opening of the loans screen.
type Session struct {
    At      time.Time
    Member  string
    Version string
    Err     string // what the screen failed with; "" if it showed the loans
}
```

```go
// Week replays the loans screen from Monday 2 to Sunday 8 November 2026:
// 4 000 members with a loan open it about once a day. The version 3.4, the
// first that knows "late", is published on Wednesday 4 and installed by
// more members each day.
func Week(answer func(loans []Loan, version string) []byte) []Session {
    rnd := rand.New(rand.NewPCG(4, 64))
    type member struct {
        name, version string
        late          bool
        updates       time.Time // when this member installs 3.4
    }
    release := time.Date(2026, 11, 4, 10, 0, 0, 0, time.UTC)
    var members []member
    for i := range 4000 {
        v := []string{"3.1", "3.2", "3.3"}[rnd.IntN(3)]
        updates := release.Add(time.Duration(rnd.ExpFloat64()*4*24) * time.Hour)
        members = append(members, member{fmt.Sprintf("m%04d", i), v, rnd.Float64() < 0.12, updates})
    }
    var out []Session
    start := time.Date(2026, 11, 2, 0, 0, 0, 0, time.UTC)
    for day := range 7 {
        for _, m := range members {
            if rnd.Float64() > 0.9 {
                continue // not today
            }
            at := start.AddDate(0, 0, day).Add(time.Duration(7*3600+rnd.IntN(14*3600)) * time.Second)
            version, read := m.version, Read(ReadBefore34)
            if at.After(m.updates) {
                version, read = "3.4", Read34
            }
            loans := []Loan{{1, "open"}}
            if m.late && at.After(LateSince) {
                loans[0].Status = "late"
            }
            s := Session{At: at, Member: m.name, Version: version}
            if _, err := read(answer(loans, version)); err != nil {
                s.Err = err.Error()
            }
            out = append(out, s)
        }
    }
    return out
}
```
